# 十五、把原Baseline迁移成V0 Shared Kernel

创建：

```bash
cat > ops/cuda/rmsnorm/src/rmsnorm_shared.cu <<'EOF'
#include "hqsb/cuda/rmsnorm.cuh"

namespace hqsb::cuda_ops {

__global__ void rmsnorm_shared_kernel(
    const float* input,
    const float* weight,
    float* output,
    int hidden,
    float epsilon)
{
    extern __shared__ float shared_sum[];

    const int row = static_cast<int>(blockIdx.x);
    const int tid = static_cast<int>(threadIdx.x);

    const std::size_t base =
        static_cast<std::size_t>(row) *
        static_cast<std::size_t>(hidden);

    float local_sum = 0.0f;

    for (int column = tid;
         column < hidden;
         column += blockDim.x)
    {
        const float value =
            input[base + column];

        local_sum += value * value;
    }

    shared_sum[tid] = local_sum;

    __syncthreads();

    for (unsigned int stride =
             blockDim.x / 2;
         stride > 0;
         stride >>= 1)
    {
        if (tid < static_cast<int>(stride)) {
            shared_sum[tid] +=
                shared_sum[
                    tid +
                    static_cast<int>(stride)
                ];
        }

        __syncthreads();
    }

    const float inverse_rms =
        rsqrtf(
            shared_sum[0] /
            static_cast<float>(hidden)
            + epsilon
        );

    for (int column = tid;
         column < hidden;
         column += blockDim.x)
    {
        output[base + column] =
            input[base + column] *
            inverse_rms *
            weight[column];
    }
}

cudaError_t launch_rmsnorm_shared_f32(
    const float* input,
    const float* weight,
    float* output,
    const RmsNormConfig& config)
{
    const std::size_t shared_bytes =
        static_cast<std::size_t>(
            config.threads) *
        sizeof(float);

    rmsnorm_shared_kernel<<<
        config.rows,
        config.threads,
        shared_bytes,
        config.stream
    >>>(
        input,
        weight,
        output,
        config.hidden,
        config.epsilon
    );

    return cudaGetLastError();
}

}  // namespace hqsb::cuda_ops
EOF
```

这里特别注意：

这个函数不允许：


# 十六、实现V1：Warp Shuffle

创建：

```bash
cat > ops/cuda/rmsnorm/src/rmsnorm_warp.cu <<'EOF'
#include "hqsb/cuda/rmsnorm.cuh"

namespace hqsb::cuda_ops {

__device__ __forceinline__
float warp_reduce_sum(float value)
{
    constexpr unsigned int mask =
        0xffffffffu;

    for (int offset = 16;
         offset > 0;
         offset >>= 1)
    {
        value +=
            __shfl_down_sync(
                mask,
                value,
                offset
            );
    }

    return value;
}

template<int BLOCK_SIZE>
__global__ void rmsnorm_warp_kernel(
    const float* input,
    const float* weight,
    float* output,
    int hidden,
    float epsilon)
{
    constexpr int kWarpSize = 32;
    constexpr int kWarpCount =
        BLOCK_SIZE / kWarpSize;

    __shared__ float warp_sums[kWarpCount];
    __shared__ float inverse_rms_shared;

    const int row =
        static_cast<int>(blockIdx.x);

    const int tid =
        static_cast<int>(threadIdx.x);

    const int lane =
        tid & (kWarpSize - 1);

    const int warp_id =
        tid / kWarpSize;

    const std::size_t base =
        static_cast<std::size_t>(row) *
        static_cast<std::size_t>(hidden);

    float local_sum = 0.0f;

    for (int column = tid;
         column < hidden;
         column += BLOCK_SIZE)
    {
        const float value =
            input[base + column];

        local_sum += value * value;
    }

    local_sum =
        warp_reduce_sum(local_sum);

    if (lane == 0) {
        warp_sums[warp_id] =
            local_sum;
    }

    __syncthreads();

    if (warp_id == 0) {

        float block_sum =
            lane < kWarpCount
            ? warp_sums[lane]
            : 0.0f;

        block_sum =
            warp_reduce_sum(block_sum);

        if (lane == 0) {
            inverse_rms_shared =
                rsqrtf(
                    block_sum /
                    static_cast<float>(
                        hidden)
                    + epsilon
                );
        }
    }

    __syncthreads();

    const float inverse_rms =
        inverse_rms_shared;

    for (int column = tid;
         column < hidden;
         column += BLOCK_SIZE)
    {
        output[base + column] =
            input[base + column] *
            inverse_rms *
            weight[column];
    }
}

template<int BLOCK_SIZE>
cudaError_t launch_impl(
    const float* input,
    const float* weight,
    float* output,
    const RmsNormConfig& config)
{
    rmsnorm_warp_kernel<
        BLOCK_SIZE
    ><<<
        config.rows,
        BLOCK_SIZE,
        0,
        config.stream
    >>>(
        input,
        weight,
        output,
        config.hidden,
        config.epsilon
    );

    return cudaGetLastError();
}

cudaError_t launch_rmsnorm_warp_f32(
    const float* input,
    const float* weight,
    float* output,
    const RmsNormConfig& config)
{
    switch (config.threads) {

        case 128:
            return launch_impl<128>(
                input,
                weight,
                output,
                config);

        case 256:
            return launch_impl<256>(
                input,
                weight,
                output,
                config);

        case 512:
            return launch_impl<512>(
                input,
                weight,
                output,
                config);

        default:
            return
                cudaErrorInvalidConfiguration;
    }
}

}  // namespace hqsb::cuda_ops
EOF
```

```bash
        return cudaErrorInvalidValue;
    }

    if (config.threads != 128 &&
        config.threads != 256 &&
        config.threads != 512)
    {
        return
            cudaErrorInvalidConfiguration;
    }

    return cudaSuccess;
}

cudaError_t rmsnorm_forward_f32(
    const float* input,
    const float* weight,
    float* output,
    const RmsNormConfig& config,
    RmsNormVariant variant)
{
    const cudaError_t validation =
        validate_config(config);

    if (validation != cudaSuccess) {
        return validation;
    }

    switch (variant) {

        case RmsNormVariant::kShared:

            return
                launch_rmsnorm_shared_f32(
                    input,
                    weight,
                    output,
                    config);

        case RmsNormVariant::kWarp:

            return
                launch_rmsnorm_warp_f32(
                    input,
                    weight,
                    output,
                    config);

        case RmsNormVariant::kAuto:

            // Temporary policy.
            // Will be replaced by measured
            // dispatch rules in later milestone.
            return
                launch_rmsnorm_warp_f32(
                    input,
                    weight,
                    output,
                    config);

        default:

            return cudaErrorNotSupported;
    }
}

}  // namespace hqsb::cuda_ops
EOF
```

## 当前最重要的一条原则

从这一阶段开始，不再按照：

> **“我还能学什么技术？”**

驱动项目。

而按照：

> **“当前测量结果暴露了什么瓶颈？下一项技术能否解决这个瓶颈？”**

驱动项目。

例如：

```text
Profiler:
Barrier Stall高
       ↓
Warp Shuffle

Profiler:
DRAM Bound
       ↓
Vectorized Load

Profiler:
Bandwidth / storage压力
       ↓
FP16 / half2

模型端到端:
Kernel launch多
       ↓
Fusion

多GPU:
通信占比高
       ↓
NCCL/Overlap

图中相邻算子反复出现
       ↓
Compiler Fusion
```

这条因果链最终会成为 **HeteroQuantServeBench 最有含金量的地方**。它会让你的CUDA、CANN、NCCL和Compiler不再是简历上的四个技术名词，而是同一个性能工程问题在不同层次上的自然解法。
